# App A: POST /demo/sin-proteccion
req = urllib.request.Request(url, datos, ...)
respuesta = json.loads(urllib.request.urlopen(req).read())
#           ↑ sin tiempo máximo: espera lo que haga falta
ia = json.loads(respuesta["candidates"][0]...)
return {"gravedad": ia["gravedad"], ...}
#           ↑ confía en lo que llegó; si algo falla: 500
# App B: POST /demo/con-proteccion
ia = preguntar_a_la_ia(INSTRUCCION + texto)
#   dentro: urlopen(req, timeout=5)      → 504 si tarda
#           except HTTPError         → 502 si se cayó
#           except ValueError...     → 502 si no es JSON
if ia.get("gravedad") not in ("alta", "media", "baja"):
    raise HTTPException(502, "La IA respondió algo inesperado.")
return {"gravedad": ia["gravedad"], ...}
